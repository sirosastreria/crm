} } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { try { if (e.parameter.action === 'version') { return jsonResponse({ ok: true, data: { version: VERSION, zona: ZONA_HORARIA } }); break; case 'schedule': result = scheduleCall(body.id, body.date, body.time, body.duration, body.sucursal); break; case 'cancelSchedule': result = cancelSchedule(body.id); break; default: return jsonResponse({ ok: false, error: 'Acción desconocida: ' + body.action }); const headers = range[0]; const idIdx = headers.indexOf('ID'); const fechaAltaIdx = headers.indexOf('Fecha Alta'); const rows = range.slice(1).filter(function (row) { return row[idIdx]; }); rows.sort(function (a, b) { return new Date(b[fechaAltaIdx]) - new Date(a[fechaAltaIdx]);
